cmake_minimum_required(VERSION 3.20)
project(QuestBridge VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_library(questbridge INTERFACE)
target_include_directories(questbridge INTERFACE "${CMAKE_CURRENT_SOURCE_DIR}/common" "${CMAKE_CURRENT_SOURCE_DIR}")
add_executable(qb_observer_tests tests/unit/observer.cpp ps5/host/observer/observer.c)
target_link_libraries(qb_observer_tests PRIVATE questbridge)
add_executable(qb_render_tests tests/unit/render.cpp)
target_link_libraries(qb_render_tests PRIVATE questbridge)
add_executable(qb_native_tests tests/unit/native.cpp)
target_link_libraries(qb_native_tests PRIVATE questbridge)
add_executable(qb_import_scan tools/diagnostics/import-scan.cpp)
target_link_libraries(qb_import_scan PRIVATE questbridge)
add_executable(qb_import_tests tests/unit/elf_imports.cpp)
target_link_libraries(qb_import_tests PRIVATE questbridge)
foreach(target qb_native_tests qb_import_scan qb_import_tests qb_render_tests qb_observer_tests)
 if(MSVC)
  target_compile_options(${target} PRIVATE /W4 /WX)
 else()
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
 endif()
endforeach()
enable_testing()
add_test(NAME render_view_projection COMMAND qb_render_tests)
add_test(NAME observer_user_only_contract COMMAND qb_observer_tests)
add_test(NAME native_models_wire_calibration_video_firmware COMMAND qb_native_tests)

add_test(NAME offline_elf_imports COMMAND qb_import_tests)

option(QB_BUILD_NATIVE_RECEIVER "Build native authenticated transport with OpenSSL 3" ON)
if(QB_BUILD_NATIVE_RECEIVER)
 find_package(OpenSSL 3 REQUIRED COMPONENTS Crypto)
 find_package(Threads REQUIRED)
 add_library(qb_transport INTERFACE)
 target_link_libraries(qb_transport INTERFACE questbridge OpenSSL::Crypto Threads::Threads)
 add_executable(qb_transport_tests tests/unit/transport.cpp)
 target_link_libraries(qb_transport_tests PRIVATE qb_transport)
 if(NOT WIN32)
  add_executable(qb_native_receiver ps5/host/native/main.cpp)
  target_link_libraries(qb_native_receiver PRIVATE qb_transport)
  find_program(QB_NODE_EXECUTABLE node)
  if(QB_NODE_EXECUTABLE)
   add_test(NAME native_node_udp COMMAND ${QB_NODE_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/integration/native-receiver.mjs $<TARGET_FILE:qb_native_receiver>)
   set_tests_properties(native_node_udp PROPERTIES TIMEOUT 30)
  endif()
 endif()
 foreach(target qb_transport_tests qb_native_receiver)
  if(TARGET ${target})
   if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
   else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
   endif()
  endif()
 endforeach()
 add_test(NAME authenticated_native_transport COMMAND qb_transport_tests)
endif()
